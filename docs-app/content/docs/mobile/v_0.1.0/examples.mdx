# Примеры использования @plainerV/ui-native

## Базовый пример

```typescript
import React, { useState } from 'react';
import { View, ScrollView } from 'react-native';
import { Button, Input, Typography, ThemeProvider } from '@plainerV/ui-native';
import { ButtonVariant, Size } from '@plainerV/ui-core';
import { lightButtonTheme, lightColors } from '@plainerV/ui-themes';

function App() {
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');

  const theme = {
    mode: 'LIGHT',
    colors: lightColors,
    buttons: lightButtonTheme,
  };

  return (
    <ThemeProvider initialTheme={theme}>
      <ScrollView style={{ padding: 20 }}>
        <Typography variant="h1" style={{ marginBottom: 20 }}>
          Добро пожаловать
        </Typography>

        <Input
          label="Email"
          placeholder="Введите email"
          value={email}
          onChangeText={setEmail}
          size={Size.MD}
        />

        <Input
          label="Пароль"
          placeholder="Введите пароль"
          value={password}
          onChangeText={setPassword}
          secureTextEntry
          size={Size.MD}
        />

        <Button
          variant={ButtonVariant.PRIMARY}
          size={Size.LG}
          fullWidth
          onPress={() => console.log('Вход')}
          style={{ marginTop: 20 }}
        >
          Войти
        </Button>
      </ScrollView>
    </ThemeProvider>
  );
}
```

## Форма с валидацией

```typescript
import React, { useState } from 'react';
import { View } from 'react-native';
import { Button, Input, Typography, ThemeProvider } from '@plainerV/ui-native';
import { ButtonVariant, Size } from '@plainerV/ui-core';

function LoginForm() {
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [errors, setErrors] = useState({ email: '', password: '' });
  const [loading, setLoading] = useState(false);

  const validate = () => {
    const newErrors = { email: '', password: '' };

    if (!email) {
      newErrors.email = 'Email обязателен';
    } else if (!email.includes('@')) {
      newErrors.email = 'Неверный формат email';
    }

    if (!password) {
      newErrors.password = 'Пароль обязателен';
    } else if (password.length < 8) {
      newErrors.password = 'Пароль должен быть минимум 8 символов';
    }

    setErrors(newErrors);
    return !newErrors.email && !newErrors.password;
  };

  const handleSubmit = async () => {
    if (validate()) {
      setLoading(true);
      // Логика входа
      setTimeout(() => {
        setLoading(false);
      }, 2000);
    }
  };

  return (
    <View style={{ padding: 20 }}>
      <Typography variant="h2" style={{ marginBottom: 30 }}>
        Вход в систему
      </Typography>

      <Input
        label="Email"
        placeholder="Введите email"
        value={email}
        onChangeText={setEmail}
        error={errors.email}
        required
        size={Size.MD}
      />

      <Input
        label="Пароль"
        placeholder="Введите пароль"
        value={password}
        onChangeText={setPassword}
        secureTextEntry
        error={errors.password}
        helperText="Минимум 8 символов"
        required
        size={Size.MD}
      />

      <Button
        variant={ButtonVariant.PRIMARY}
        size={Size.LG}
        fullWidth
        loading={loading}
        onPress={handleSubmit}
        style={{ marginTop: 20 }}
      >
        Войти
      </Button>
    </View>
  );
}
```

## Переключение темы

```typescript
import React, { useState } from 'react';
import { View } from 'react-native';
import { Button, ThemeProvider, useTheme, ThemeMode } from '@plainerV/ui-native';
import { lightButtonTheme, lightColors, darkButtonTheme, darkColors } from '@plainerV/ui-themes';

function App() {
  const [isDark, setIsDark] = useState(false);

  const theme = {
    mode: isDark ? ThemeMode.DARK : ThemeMode.LIGHT,
    colors: isDark ? darkColors : lightColors,
    buttons: isDark ? darkButtonTheme : lightButtonTheme,
  };

  return (
    <ThemeProvider initialTheme={theme}>
      <View style={{ flex: 1, padding: 20 }}>
        <ThemeToggleButton isDark={isDark} onToggle={() => setIsDark(!isDark)} />
        {/* Остальные компоненты */}
      </View>
    </ThemeProvider>
  );
}

function ThemeToggleButton({ isDark, onToggle }: { isDark: boolean; onToggle: () => void }) {
  const { toggle } = useTheme();

  return (
    <Button
      variant={isDark ? ButtonVariant.SECONDARY : ButtonVariant.PRIMARY}
      onPress={() => {
        toggle();
        onToggle();
      }}
    >
      {isDark ? 'Светлая тема' : 'Темная тема'}
    </Button>
  );
}
```

## Комплексный пример формы

```typescript
import React, { useState } from 'react';
import { View, ScrollView, KeyboardAvoidingView, Platform } from 'react-native';
import { Button, Input, Typography, ThemeProvider } from '@plainerV/ui-native';
import { ButtonVariant, Size } from '@plainerV/ui-core';
import { lightButtonTheme, lightColors } from '@plainerV/ui-themes';

function RegistrationForm() {
  const [formData, setFormData] = useState({
    firstName: '',
    lastName: '',
    email: '',
    phone: '',
    password: '',
    confirmPassword: '',
  });

  const [errors, setErrors] = useState<Record<string, string>>({});

  const handleChange = (field: string, value: string) => {
    setFormData(prev => ({ ...prev, [field]: value }));
    // Очищаем ошибку при изменении
    if (errors[field]) {
      setErrors(prev => ({ ...prev, [field]: '' }));
    }
  };

  const validate = () => {
    const newErrors: Record<string, string> = {};

    if (!formData.firstName) newErrors.firstName = 'Имя обязательно';
    if (!formData.lastName) newErrors.lastName = 'Фамилия обязательна';
    if (!formData.email) {
      newErrors.email = 'Email обязателен';
    } else if (!/\S+@\S+\.\S+/.test(formData.email)) {
      newErrors.email = 'Неверный формат email';
    }
    if (!formData.password) {
      newErrors.password = 'Пароль обязателен';
    } else if (formData.password.length < 8) {
      newErrors.password = 'Пароль должен быть минимум 8 символов';
    }
    if (formData.password !== formData.confirmPassword) {
      newErrors.confirmPassword = 'Пароли не совпадают';
    }

    setErrors(newErrors);
    return Object.keys(newErrors).length === 0;
  };

  const handleSubmit = () => {
    if (validate()) {
      console.log('Форма отправлена', formData);
    }
  };

  const theme = {
    mode: 'LIGHT',
    colors: lightColors,
    buttons: lightButtonTheme,
  };

  return (
    <ThemeProvider initialTheme={theme}>
      <KeyboardAvoidingView
        behavior={Platform.OS === 'ios' ? 'padding' : 'height'}
        style={{ flex: 1 }}
      >
        <ScrollView style={{ flex: 1 }} contentContainerStyle={{ padding: 20 }}>
          <Typography variant="h1" style={{ marginBottom: 30 }}>
            Регистрация
          </Typography>

          <Input
            label="Имя"
            placeholder="Введите имя"
            value={formData.firstName}
            onChangeText={value => handleChange('firstName', value)}
            error={errors.firstName}
            required
            size={Size.MD}
          />

          <Input
            label="Фамилия"
            placeholder="Введите фамилию"
            value={formData.lastName}
            onChangeText={value => handleChange('lastName', value)}
            error={errors.lastName}
            required
            size={Size.MD}
          />

          <Input
            label="Email"
            placeholder="Введите email"
            value={formData.email}
            onChangeText={value => handleChange('email', value)}
            error={errors.email}
            keyboardType="email-address"
            autoCapitalize="none"
            required
            size={Size.MD}
          />

          <Input
            label="Телефон"
            placeholder="+7 (999) 999-99-99"
            value={formData.phone}
            onChangeText={value => handleChange('phone', value)}
            keyboardType="phone-pad"
            size={Size.MD}
          />

          <Input
            label="Пароль"
            placeholder="Введите пароль"
            value={formData.password}
            onChangeText={value => handleChange('password', value)}
            error={errors.password}
            secureTextEntry
            helperText="Минимум 8 символов"
            required
            size={Size.MD}
          />

          <Input
            label="Подтвердите пароль"
            placeholder="Повторите пароль"
            value={formData.confirmPassword}
            onChangeText={value => handleChange('confirmPassword', value)}
            error={errors.confirmPassword}
            secureTextEntry
            required
            size={Size.MD}
          />

          <Button
            variant={ButtonVariant.PRIMARY}
            size={Size.LG}
            fullWidth
            onPress={handleSubmit}
            style={{ marginTop: 20 }}
          >
            Зарегистрироваться
          </Button>
        </ScrollView>
      </KeyboardAvoidingView>
    </ThemeProvider>
  );
}
```
